% Opening remarks: the goal of the challenge (wording from index.html #about)
\begin{frame}{The Goal of the Challenge}
  \begin{center}
    {\small\color{charalMuted}CHARAL 2026 is held within \hl{MICAI 2026}, the Mexican International Conference
    on Artificial Intelligence.}
  \end{center}
  \vspace{1mm}
  \begin{columns}[T, onlytextwidth]
    \column{0.315\textwidth}
    \begin{charalcard}{The problem}
      Generative AI is easy to adopt \hl{passively}, as a shortcut that does the thinking for the learner.
    \end{charalcard}
    \column{0.315\textwidth}
    \begin{charalcard}{Our goal}
      AI systems that \hl{interact with, audit and evaluate} the learner or researcher, so that critical
      thinking and deep reasoning actively increase.
    \end{charalcard}
    \column{0.315\textwidth}
    \begin{charalcard}{The shift}
      From \hl{detecting AI use} to \hl{demanding higher-order thinking} when AI is used, with rigorous
      engineering and data governance.
    \end{charalcard}
  \end{columns}
  \vspace{4mm}
  \begin{center}
    {\headingfont\normalsize\color{white}Cognition First: AI that elevates human thinking, not replaces it.}
  \end{center}
\end{frame}
