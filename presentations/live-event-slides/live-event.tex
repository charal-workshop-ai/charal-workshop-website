% Circular photo that always fills the circle:
%   \charalface{file}{diameter}{includegraphics opts}{x offset}{y offset}
\newcommand{\charalface}[5]{%
  \tikz[baseline=(c.center)]{
    \begin{scope}
      \clip (0,0) circle (#2/2);
      \node[anchor=center, inner sep=0] at (#4,#5) {\includegraphics[#3]{#1}};
    \end{scope}
    \draw[charalMint, line width=0.8pt] (0,0) circle (#2/2);
    \coordinate (c) at (0,0);}}

% Rounded portrait: \charalportrait{file}{width}{height}{includegraphics opts}{x offset}{y offset}
\newcommand{\charalportrait}[6]{%
  \tikz[baseline=(c.center)]{
    \begin{scope}
      \clip[rounded corners=3mm] (-#2/2,-#3/2) rectangle (#2/2,#3/2);
      \node[anchor=center, inner sep=0] at (#5,#6) {\includegraphics[#4]{#1}};
    \end{scope}
    \draw[charalMint, line width=0.8pt, rounded corners=3mm] (-#2/2,-#3/2) rectangle (#2/2,#3/2);
    \coordinate (c) at (0,0);}}
