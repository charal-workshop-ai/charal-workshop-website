\newcommand{\critrow}[2]{% {percent}{criterion}
  \noindent\makebox[35mm][l]{#2}%
  \tikz[baseline=-0.7ex]{\fill[charalMint!20!charalCard] (0,-0.9mm) rectangle (16mm,0.9mm);
                         \fill[charalMint] (0,-0.9mm) rectangle (#1*0.64mm,0.9mm);}%
  \hfill{\color{charalMint}\bfseries #1\%}\par\vskip1.4mm}
\newcommand{\deliv}[2]{% {num}{text}
  \noindent\charalnum{#1}\hspace{2mm}\begin{minipage}[t]{\dimexpr\linewidth-8mm\relax}\raggedright #2\end{minipage}\par\vskip1.6mm}
